if ( ! function_exists( 'infosystem_wc_cursos_url' ) ) {
	function infosystem_wc_cursos_url() {
		$t = get_term_by( 'slug', 'cursos-subvencionados-castilla-la-mancha', 'product_cat' );
		if ( $t && ! is_wp_error( $t ) ) {
			$l = get_term_link( $t );
			if ( ! is_wp_error( $l ) ) {
				return $l;
			}
		}
		return home_url( '/cursos-subvencionados-castilla-la-mancha/' );
	}
}

add_action( 'init', function () {
	$slug = 'como-funcionan-cursos-subvencionados-sepe-castilla-la-mancha';
	$opt  = 'infosystem_como_funciona_page_id';
	$id   = (int) get_option( $opt );
	if ( $id && 'publish' === get_post_status( $id ) ) {
		return;
	}
	$p = get_page_by_path( $slug, OBJECT, 'page' );
	if ( $p instanceof WP_Post ) {
		update_option( $opt, $p->ID );
		return;
	}
	$nid = wp_insert_post( array( 'post_title' => 'Cómo funcionan los cursos subvencionados por el SEPE en Castilla-La Mancha', 'post_name' => $slug, 'post_status' => 'publish', 'post_type' => 'page', 'post_content' => infosystem_como_funciona_html() ), true );
	if ( ! is_wp_error( $nid ) && $nid ) {
		update_option( $opt, $nid );
		update_post_meta( $nid, 'rank_math_title', 'Cómo funcionan los cursos subvencionados SEPE en Castilla-La Mancha | Infosystem' );
		update_post_meta( $nid, 'rank_math_description', 'Requisitos, inscripción gratuita, formación online y certificado de cursos subvencionados por el SEPE en Castilla-La Mancha.' );
	}
}, 20 );
add_filter( 'the_content', function ( $content ) {
	if ( is_page( 'como-funcionan-cursos-subvencionados-sepe-castilla-la-mancha' ) && strlen( trim( wp_strip_all_tags( $content ) ) ) < 80 ) {
		return infosystem_como_funciona_html();
	}
	return $content;
}, 5 );
add_filter( 'the_content', function ( $content ) {
	if ( ! is_page( array( 16852, 'como-funcionan-cursos-subvencionados-sepe-castilla-la-mancha' ) ) ) {
		return $content;
	}
	if ( strpos( $content, 'cf-blog-cards' ) === false ) {
		return $content;
	}
	$content = preg_replace(
		'/<a class="cf-blog-card"([^>]*)><img([^>]*?)>\s*<\/p>\s*<div class="cf-blog-card__body">/i',
		'<a class="cf-blog-card"$1><img$2><div class="cf-blog-card__body">',
		$content
	);
	$content = preg_replace(
		'/<p class="cf-blog-card__excerpt">/i',
		'<div class="cf-blog-card__excerpt">',
		$content
	);
	$content = preg_replace(
		'/<\/p>\s*(<span class="cf-blog-card__more">)/i',
		'</div>$1',
		$content
	);
	$content = preg_replace(
		'/<\/div>\s*<p><\/a>(?:<br\s*\/?>)?\s*/i',
		'</div></a>',
		$content
	);
	return $content;
}, 12 );
add_action( 'template_redirect', function () {
	if ( is_admin() || ! is_front_page() ) {
		return;
	}
	$cursos   = esc_url( infosystem_wc_cursos_url() );
	$como     = esc_url( infosystem_como_funciona_url() );
	$contacto = esc_url( home_url( '/contacto/' ) );
	$home     = esc_url( home_url( '/' ) );
	ob_start( function ( $html ) use ( $cursos, $como, $contacto, $home ) {
		if ( ! is_string( $html ) ) {
			return $html;
		}
		$html = preg_replace(
			'/(<a[^>]*class="[^"]*e-c2f467e-da956f3[^"]*"[^>]*href=")[^"]*("/iu',
			'$1' . $contacto . '$2',
			$html,
			1
		);
		$html = preg_replace(
			'/(<a[^>]*class="[^"]*tp-button-outline[^"]*"[^>]*href=")[^"]*("[^>]*>\s*Solicitar información\s*<\/a>)/iu',
			'$1' . $contacto . '$2',
			$html,
			1
		);
		$html = str_replace( 'https://1.envato.market/Yx2YR', $contacto, $html );
		$html = preg_replace( '#https?://1\.envato\.market/[^"\'<>\s]+#', $contacto, $html );
		$html = str_replace( 'href="' . $home . '" target="_self" class="tp-button-outline e-415bdea', 'href="' . $cursos . '" target="_self" class="tp-button-outline e-415bdea', $html );
		$html = str_replace( 'href="' . $home . '" target="_self" class="tp-button-outline e-5264a37', 'href="' . $contacto . '" target="_self" class="tp-button-outline e-5264a37', $html );
		$html = str_replace( 'href="' . esc_url( home_url( '/packages/' ) ) . '"', 'href="' . $cursos . '"', $html );
		$html = str_replace(
			'data-interaction-id="85bed70"> Ver todos los cursos </a>',
			'data-interaction-id="85bed70"> Ver itinerarios formativos </a>',
			$html
		);
		$html = str_replace( 'View All Packages', 'Ver itinerarios formativos', $html );
		$html = str_replace( 'href="' . esc_url( home_url( '/courses/' ) ) . '"', 'href="' . $cursos . '"', $html );
		$html = str_replace( 'View All</a>', 'Ver todos los cursos</a>', $html );
		$html = str_replace( 'Cursos completamente actualizados.', 'Aprende a tu ritmo. Cursos online actualizados, sin horarios fijos.', $html );
		$html = preg_replace( '/<button(\s+class="tp-button-primary e-100c151[^"]*"[^>]*)>\s*Cómo funciona\s*<\/button>/iu', '<a$1 href="' . $como . '">Cómo funciona</a>', $html, 1 );
		return $html;
	} );
}, 0 );
